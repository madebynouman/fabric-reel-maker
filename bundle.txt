package com.madebynouman.fabricreel

import android.content.Context
import android.media.AudioFormat
import android.media.MediaCodec
import android.media.MediaCodecInfo
import android.media.MediaExtractor
import android.media.MediaFormat
import android.net.Uri
import java.nio.ByteOrder
import kotlin.math.min
import kotlin.math.roundToInt

class AudioSample(val data: ByteArray, val ptsUs: Long, val flags: Int)
class EncodedAudio(val format: MediaFormat, val samples: List<AudioSample>)

/** Decodes any music file the phone can play, trims or loops it to the video length, fades it, and encodes AAC. */
object AudioMaker {

    fun encodeMusic(ctx: Context, uri: Uri, durationUs: Long): EncodedAudio {
        val (pcm, rate, ch) = decode(ctx, uri, durationUs)
        val shaped = shape(pcm, rate, ch, durationUs)
        return encodeAac(shaped, rate, min(ch, 2))
    }

    private class ShortList {
        var data = ShortArray(1 shl 16)
        var size = 0
        fun ensure(extra: Int) {
            if (size + extra > data.size) {
                var n = data.size
                while (size + extra > n) n *= 2
                data = data.copyOf(n)
            }
        }
    }

    private fun decode(ctx: Context, uri: Uri, durationUs: Long): Triple<ShortArray, Int, Int> {
        val ex = MediaExtractor()
        ex.setDataSource(ctx, uri, null)
        var track = -1
        for (i in 0 until ex.trackCount) {
            val mime = ex.getTrackFormat(i).getString(MediaFormat.KEY_MIME) ?: continue
            if (mime.startsWith("audio/")) { track = i; break }
        }
        require(track >= 0) { "No audio in this file" }
        ex.selectTrack(track)
        val inFmt = ex.getTrackFormat(track)
        val dec = MediaCodec.createDecoderByType(inFmt.getString(MediaFormat.KEY_MIME)!!)
        dec.configure(inFmt, null, null, 0)
        dec.start()

        var rate = inFmt.getInteger(MediaFormat.KEY_SAMPLE_RATE)
        var ch = inFmt.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
        var isFloat = false
        val out = ShortList()
        val info = MediaCodec.BufferInfo()
        var inputDone = false
        try {
            while (true) {
                if (!inputDone) {
                    val ii = dec.dequeueInputBuffer(10_000)
                    if (ii >= 0) {
                        val buf = dec.getInputBuffer(ii)!!
                        val size = ex.readSampleData(buf, 0)
                        if (size < 0) {
                            dec.queueInputBuffer(ii, 0, 0, 0, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                            inputDone = true
                        } else {
                            dec.queueInputBuffer(ii, 0, size, ex.sampleTime, 0)
                            ex.advance()
                        }
                    }
                }
                val oi = dec.dequeueOutputBuffer(info, 10_000)
                if (oi == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
                    val f = dec.outputFormat
                    rate = f.getInteger(MediaFormat.KEY_SAMPLE_RATE)
                    ch = f.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
                    isFloat = f.containsKey(MediaFormat.KEY_PCM_ENCODING) &&
                        f.getInteger(MediaFormat.KEY_PCM_ENCODING) == AudioFormat.ENCODING_PCM_FLOAT
                } else if (oi >= 0) {
                    val buf = dec.getOutputBuffer(oi)
                    if (buf != null && info.size > 0) {
                        buf.position(info.offset)
                        buf.limit(info.offset + info.size)
                        buf.order(ByteOrder.nativeOrder())
                        if (isFloat) {
                            val fb = buf.asFloatBuffer()
                            val n = fb.remaining()
                            out.ensure(n)
                            for (k in 0 until n) {
                                out.data[out.size++] = (fb.get().coerceIn(-1f, 1f) * 32767f).roundToInt().toShort()
                            }
                        } else {
                            val sb = buf.asShortBuffer()
                            val n = sb.remaining()
                            out.ensure(n)
                            sb.get(out.data, out.size, n)
                            out.size += n
                        }
                    }
                    dec.releaseOutputBuffer(oi, false)
                    if ((info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) break
                    val needSamples = durationUs * rate / 1_000_000L * ch
                    if (out.size >= needSamples) break
                }
            }
        } finally {
            runCatching { dec.stop() }
            dec.release()
            ex.release()
        }
        require(out.size >= ch * rate / 10) { "Music file is too short" }
        return Triple(out.data.copyOf(out.size), rate, ch)
    }

    /** Stereo/mono only, loop if the song is shorter than the video, gentle fade in and a longer fade out. */
    private fun shape(src: ShortArray, rate: Int, ch: Int, durationUs: Long): ShortArray {
        val outCh = min(ch, 2)
        val frames = (durationUs * rate / 1_000_000L).toInt()
        val srcFrames = src.size / ch
        val out = ShortArray(frames * outCh)
        val fadeIn = (0.3f * rate).toInt()
        val fadeOut = (1.8f * rate).toInt()
        for (f in 0 until frames) {
            val sf = f % srcFrames
            var g = 0.9f
            if (f < fadeIn) g *= f.toFloat() / fadeIn
            val left = frames - f
            if (left < fadeOut) g *= left.toFloat() / fadeOut
            for (c in 0 until outCh) {
                out[f * outCh + c] = (src[sf * ch + c] * g).roundToInt().coerceIn(-32768, 32767).toShort()
            }
        }
        return out
    }

    private fun encodeAac(pcm: ShortArray, rate: Int, ch: Int): EncodedAudio {
        val fmt = MediaFormat.createAudioFormat(MediaFormat.MIMETYPE_AUDIO_AAC, rate, ch).apply {
            setInteger(MediaFormat.KEY_AAC_PROFILE, MediaCodecInfo.CodecProfileLevel.AACObjectLC)
            setInteger(MediaFormat.KEY_BIT_RATE, 192_000)
            setInteger(MediaFormat.KEY_MAX_INPUT_SIZE, 16_384)
        }
        val enc = MediaCodec.createEncoderByType(MediaFormat.MIMETYPE_AUDIO_AAC)
        enc.configure(fmt, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE)
        enc.start()
        val samples = ArrayList<AudioSample>()
        var outFormat: MediaFormat? = null
        val info = MediaCodec.BufferInfo()
        var pos = 0
        var inputDone = false
        try {
            while (true) {
                if (!inputDone) {
                    val ii = enc.dequeueInputBuffer(10_000)
                    if (ii >= 0) {
                        val buf = enc.getInputBuffer(ii)!!
                        buf.clear()
                        var n = min(buf.capacity() / 2, pcm.size - pos)
                        n -= n % ch
                        val pts = (pos / ch).toLong() * 1_000_000L / rate
                        if (n <= 0) {
                            enc.queueInputBuffer(ii, 0, 0, pts, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                            inputDone = true
                        } else {
                            buf.order(ByteOrder.nativeOrder()).asShortBuffer().put(pcm, pos, n)
                            enc.queueInputBuffer(ii, 0, n * 2, pts, 0)
                            pos += n
                        }
                    }
                }
                val oi = enc.dequeueOutputBuffer(info, 10_000)
                if (oi == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
                    outFormat = enc.outputFormat
                } else if (oi >= 0) {
                    val buf = enc.getOutputBuffer(oi)
                    val isConfig = (info.flags and MediaCodec.BUFFER_FLAG_CODEC_CONFIG) != 0
                    if (!isConfig && info.size > 0 && buf != null) {
                        val bytes = ByteArray(info.size)
                        buf.position(info.offset)
                        buf.get(bytes)
                        samples.add(AudioSample(bytes, info.presentationTimeUs, info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM.inv()))
                    }
                    enc.releaseOutputBuffer(oi, false)
                    if ((info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) break
                }
            }
        } finally {
            runCatching { enc.stop() }
            enc.release()
        }
        return EncodedAudio(outFormat ?: error("AAC encoder gave no format"), samples)
    }
}

import android.graphics.Bitmap
import android.media.MediaMetadataRetriever
import java.io.File

/** Reads the frames of an AI clip a few at a time, in order. */
class ClipReader(file: File) {
    private val mmr = MediaMetadataRetriever().apply { setDataSource(file.absolutePath) }
    val frameCount: Int
    val durationSec: Float
    private val fps: Float
    private var batchStart = -1
    private var batch: List<Bitmap> = emptyList()
    private val params = MediaMetadataRetriever.BitmapParams().apply { preferredConfig = Bitmap.Config.ARGB_8888 }

    init {
        val ms = mmr.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION)?.toLongOrNull() ?: 5000L
        durationSec = ms / 1000f
        frameCount = mmr.extractMetadata(MediaMetadataRetriever.METADATA_KEY_VIDEO_FRAME_COUNT)?.toIntOrNull()
            ?: (durationSec * 24).toInt()
        fps = if (durationSec > 0f) frameCount / durationSec else 24f
    }

    fun frameAt(t: Float): Bitmap {
        val idx = (t * fps).toInt().coerceIn(0, frameCount - 1)
        if (batchStart < 0 || idx < batchStart || idx >= batchStart + batch.size) {
            batch.forEach { it.recycle() }
            val count = minOf(BATCH, frameCount - idx)
            batch = try {
                mmr.getFramesAtIndex(idx, count, params)
            } catch (e: Exception) {
                val one = mmr.getFrameAtTime((t * 1_000_000).toLong(), MediaMetadataRetriever.OPTION_CLOSEST)
                    ?: throw IllegalStateException("Couldn't read a frame of the AI clip")
                listOf(one)
            }
            batchStart = idx
        }
        return batch[idx - batchStart]
    }

    fun release() {
        batch.forEach { it.recycle() }
        batch = emptyList()
        runCatching { mmr.release() }
    }

    companion object {
        private const val BATCH = 6

        fun durationOf(file: File): Float {
            val m = MediaMetadataRetriever()
            return try {
                m.setDataSource(file.absolutePath)
                (m.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION)?.toLongOrNull() ?: 5000L) / 1000f
            } finally {
                runCatching { m.release() }
            }
        }
    }
}

import android.content.Context

/**
 * The closing card. Each line of the card (logo, text rows, button, website...) is found
 * automatically and revealed one after another, like a designed end card.
 */
class EndCard private constructor(
    private val bmp: Bitmap,
    private val bg: Int,
    private val bands: List<RectF>,
    private val W: Int,
    private val H: Int,
) {
    private val base = Matrix()
    private val m = Matrix()
    private val paint = Paint(Paint.ANTI_ALIAS_FLAG or Paint.FILTER_BITMAP_FLAG)
    private val cover = Paint().apply { color = bg }
    private val u = W / 1080f

    init {
        // Cover the frame with the card; if that would cut off its content (e.g. a 9:16 card in a
        // 4:5 video), shrink it so all content fits and fill the rest with the card colour.
        val cw = bmp.width.toFloat()
        val ch = bmp.height.toFloat()
        var s = max(W / cw, H / ch)
        var ox = (W - cw * s) / 2
        var oy = (H - ch * s) / 2
        if (bands.isNotEmpty()) {
            val top = bands.first().top
            val bottom = bands.last().bottom
            val cTop = oy + top * s
            val cBottom = oy + bottom * s
            val contentW = contentRight - contentLeft
            val cLeft = ox + contentLeft * s
            val cRight = ox + contentRight * s
            if (cTop < H * 0.06f || cBottom > H * 0.94f || cLeft < W * 0.04f || cRight > W * 0.96f) {
                s = min(s, min(0.86f * H / (bottom - top), 0.9f * W / max(1f, contentW)))
                ox = W / 2f - (contentLeft + contentRight) / 2f * s
                oy = H / 2f - (top + bottom) / 2f * s
            }
        }
        base.setScale(s, s)
        base.postTranslate(ox, oy)
    }

    private val contentLeft: Float get() = bands.minOfOrNull { it.left } ?: 0f
    private val contentRight: Float get() = bands.maxOfOrNull { it.right } ?: bmp.width.toFloat()

    fun draw(c: Canvas, t: Float) {
        c.drawColor(bg)
        val zoom = 1f + 0.012f * min(t / 4f, 1f)
        m.set(base)
        m.postScale(zoom, zoom, W / 2f, H / 2f)
        c.drawBitmap(bmp, m, paint)
        if (bands.isEmpty()) return

        val r = RectF()
        for ((i, band) in bands.withIndex()) {
            val start = if (i == 0) 0.15f else 0.85f + 0.2f * (i - 1)
            val k = easeOut((t - start) / (if (i == 0) 0.9f else 0.6f))
            // full-width strip so the reveal never leaves a visible edge
            r.set(0f, band.top - 2f, bmp.width.toFloat(), band.bottom + 2f)
            m.mapRect(r)
            r.left = 0f; r.right = W.toFloat()
            c.drawRect(r, cover)
            if (k <= 0f) continue
            c.save()
            if (i == 0) {
                val sc = 1.035f - 0.035f * k
                c.scale(sc, sc, r.centerX(), r.centerY())
            } else {
                c.translate(0f, (1 - k) * 14f * u)
            }
            c.clipRect(r)
            paint.alpha = (k * 255).roundToInt()
            c.drawBitmap(bmp, m, paint)
            paint.alpha = 255
            c.restore()
        }
    }

    fun recycle() = bmp.recycle()

    companion object {
        fun create(ctx: Context, src: EndCardSource, W: Int, H: Int): EndCard {
            val raw = when (src) {
                is EndCardSource.Image -> decode(src.file, W)
                is EndCardSource.Text -> textCard(src, W, H)
            }
            val trimmed = trimDarkBorders(raw)
            if (trimmed != raw) raw.recycle()
            val bg = backgroundColor(trimmed)
            val bands = findBands(trimmed, bg)
            return EndCard(trimmed, bg, bands, W, H)
        }

        private fun decode(file: java.io.File, W: Int): Bitmap {
            val source = ImageDecoder.createSource(file)
            return ImageDecoder.decodeBitmap(source) { dec, info, _ ->
                dec.allocator = ImageDecoder.ALLOCATOR_SOFTWARE
                var sample = 1
                while (min(info.size.width, info.size.height) / (sample * 2) >= W) sample *= 2
                if (sample > 1) dec.setTargetSampleSize(sample)
            }.let { if (it.config != Bitmap.Config.ARGB_8888) it.copy(Bitmap.Config.ARGB_8888, false) else it }
        }

        private fun lum(p: Int) = (Color.red(p) * 299 + Color.green(p) * 587 + Color.blue(p) * 114) / 1000

        /** Screenshots often have black bars at the sides; cut them off. */
        private fun trimDarkBorders(b: Bitmap): Bitmap {
            val w = b.width
            val h = b.height
            val row = IntArray(w)
            val col = IntArray(h)
            fun colDark(x: Int): Boolean {
                b.getPixels(col, 0, 1, x, 0, 1, h)
                return col.count { lum(it) < 28 } > h * 0.9
            }
            fun rowDark(y: Int): Boolean {
                b.getPixels(row, 0, w, 0, y, w, 1)
                return row.count { lum(it) < 28 } > w * 0.9
            }
            var l = 0
            var r = w - 1
            var t = 0
            var btm = h - 1
            while (l < w / 3 && colDark(l)) l++
            while (r > w * 2 / 3 && colDark(r)) r--
            while (t < h / 3 && rowDark(t)) t++
            while (btm > h * 2 / 3 && rowDark(btm)) btm--
            if (l == 0 && t == 0 && r == w - 1 && btm == h - 1) return b
            // A card that is dark itself (not just bars at the edges) is left as it is.
            if (r - l < w * 0.5f || btm - t < h * 0.5f) return b
            // step a few pixels further in to drop the soft edge of the bar
            if (l > 0) l += 3
            if (r < w - 1) r -= 3
            if (t > 0) t += 3
            if (btm < h - 1) btm -= 3
            return Bitmap.createBitmap(b, l, t, r - l + 1, btm - t + 1)
        }

        private fun backgroundColor(b: Bitmap): Int {
            val w = b.width
            val h = b.height
            val rs = ArrayList<Int>()
            val gs = ArrayList<Int>()
            val bs = ArrayList<Int>()
            fun add(x: Int, y: Int) {
                val p = b.getPixel(x.coerceIn(0, w - 1), y.coerceIn(0, h - 1))
                rs.add(Color.red(p)); gs.add(Color.green(p)); bs.add(Color.blue(p))
            }
            val ix = (w * 0.04f).toInt()
            val iy = (h * 0.03f).toInt()
            for (k in 0 until 60) {
                val x = ix + (w - 2 * ix) * k / 59
                val y = iy + (h - 2 * iy) * k / 59
                add(x, iy); add(x, h - 1 - iy); add(ix, y); add(w - 1 - ix, y)
            }
            rs.sort(); gs.sort(); bs.sort()
            return Color.rgb(rs[rs.size / 2], gs[gs.size / 2], bs[bs.size / 2])
        }

        /** Horizontal strips that contain something different from the background colour. */
        private fun findBands(b: Bitmap, bg: Int): List<RectF> {
            val w = b.width
            val h = b.height
            val br = Color.red(bg)
            val bgG = Color.green(bg)
            val bb = Color.blue(bg)
            val row = IntArray(w)
            val rowHas = BooleanArray(h)
            val rowL = IntArray(h) { w }
            val rowR = IntArray(h) { -1 }
            val minCount = max(2, (w * 0.004f).toInt())
            for (y in 0 until h) {
                b.getPixels(row, 0, w, 0, y, w, 1)
                var count = 0
                var x = 0
                while (x < w) {
                    val p = row[x]
                    val d = abs(Color.red(p) - br) + abs(Color.green(p) - bgG) + abs(Color.blue(p) - bb)
                    if (d > 70) {
                        count++
                        if (x < rowL[y]) rowL[y] = x
                        if (x > rowR[y]) rowR[y] = x
                    }
                    x += 2
                }
                rowHas[y] = count >= minCount
            }
            val gap = max(4, (h * 0.015f).toInt())
            val bands = ArrayList<RectF>()
            var y = 0
            while (y < h) {
                if (!rowHas[y]) { y++; continue }
                var end = y
                var blank = 0
                var l = rowL[y]
                var r = rowR[y]
                var k = y + 1
                while (k < h && blank <= gap) {
                    if (rowHas[k]) {
                        end = k; blank = 0
                        l = min(l, rowL[k]); r = max(r, rowR[k])
                    } else blank++
                    k++
                }
                if (end - y >= 2) bands.add(RectF(l.toFloat(), y.toFloat(), (r + 1).toFloat(), (end + 1).toFloat()))
                y = end + 1
            }
            // Too many pieces (busy card): merge the closest ones.
            while (bands.size > 7) {
                var bi = 0
                var best = Float.MAX_VALUE
                for (i in 0 until bands.size - 1) {
                    val g = bands[i + 1].top - bands[i].bottom
                    if (g < best) { best = g; bi = i }
                }
                bands[bi].union(bands[bi + 1])
                bands.removeAt(bi + 1)
            }
            // A card filled edge to edge (e.g. a photo) has no "lines" to reveal.
            if (bands.size == 1 && bands[0].height() > h * 0.8f) return emptyList()
            return bands
        }

        /** Plain-font card built from the store details, used when no end card image is chosen. */
        private fun textCard(src: EndCardSource.Text, W: Int, H: Int): Bitmap {
            val u = W / 1080f
            val b = Bitmap.createBitmap(W, H, Bitmap.Config.ARGB_8888)
            val c = Canvas(b)
            c.drawColor(Color.rgb(246, 239, 232))
            val ink = Color.rgb(60, 40, 50)
            val soft = Color.rgb(110, 80, 95)
            val p = Paint(Paint.ANTI_ALIAS_FLAG).apply { textAlign = Paint.Align.CENTER }
            fun line(text: String, size: Float, face: Typeface, color: Int, spacing: Float, y: Float) {
                if (text.isBlank()) return
                p.typeface = face; p.color = color; p.letterSpacing = spacing; p.textSize = size
                val maxW = W * 0.86f
                if (p.measureText(text) > maxW) p.textSize = size * maxW / p.measureText(text)
                c.drawText(text, W / 2f, y, p)
            }
            val medium = Typeface.create("sans-serif-medium", Typeface.NORMAL)
            val regular = Typeface.create("sans-serif", Typeface.NORMAL)
            var y = H * 0.40f
            line(src.name, 96f * u, medium, ink, 0.02f, y)
            y += 80f * u
            line(src.tagline.uppercase(), 30f * u, medium, ink, 0.3f, y)
            y += 70f * u
            p.color = Color.rgb(180, 140, 120); p.strokeWidth = 2f * u
            c.drawLine(W / 2f - 110f * u, y, W / 2f + 110f * u, y, p)
            y += 110f * u
            if (src.whatsapp.isNotBlank()) {
                line("Order on WhatsApp  ${src.whatsapp}", 40f * u, medium, ink, 0.02f, y)
                y += 70f * u
            }
            line(src.website, 36f * u, regular, soft, 0.04f, y)
            return b
        }
    }

/** Minimal fal.ai REST client: upload a photo, queue a video job, wait, download the result. */
class FalClient(private val key: String) {

    private fun open(url: String, method: String): HttpURLConnection =
        (URL(url).openConnection() as HttpURLConnection).apply {
            requestMethod = method
            connectTimeout = 30_000
            readTimeout = 120_000
            setRequestProperty("Authorization", "Key $key")
            setRequestProperty("Accept", "application/json")
        }

    private fun readBody(c: HttpURLConnection): String {
        val stream = if (c.responseCode in 200..299) c.inputStream else (c.errorStream ?: c.inputStream)
        return stream.bufferedReader().use { it.readText() }
    }

    private fun errorText(code: Int, body: String): String {
        val detail = runCatching {
            val j = JSONObject(body)
            when (val d = j.opt("detail")) {
                is String -> d
                null -> j.optString("error", body)
                else -> d.toString()
            }
        }.getOrDefault(body)
        return when (code) {
            401, 403 -> "fal.ai refused the API key. Check the key in the app."
            402 -> "Your fal.ai balance is empty. Add credit at fal.ai and try again."
            else -> "fal.ai error $code: ${detail.take(300)}"
        }
    }

    private fun postJson(url: String, json: JSONObject): JSONObject {
        val c = open(url, "POST")
        c.doOutput = true
        c.setRequestProperty("Content-Type", "application/json")
        c.outputStream.use { it.write(json.toString().toByteArray()) }
        val code = c.responseCode
        val body = readBody(c)
        c.disconnect()
        if (code !in 200..299) throw FalException(errorText(code, body))
        return JSONObject(body)
    }

    private fun getJson(url: String): Pair<Int, JSONObject?> {
        val c = open(url, "GET")
        val code = c.responseCode
        val body = readBody(c)
        c.disconnect()
        if (code !in 200..299) throw FalException(errorText(code, body))
        return code to runCatching { JSONObject(body) }.getOrNull()
    }

    /** Puts the photo on fal's file storage and returns its URL (falls back to sending it inline). */
    fun uploadJpeg(bytes: ByteArray, name: String): String {
        for (base in listOf("https://rest.fal.ai", "https://rest.alpha.fal.ai")) {
            try {
                val init = postJson(
                    "$base/storage/upload/initiate?storage_type=fal-cdn-v3",
                    JSONObject().put("content_type", "image/jpeg").put("file_name", name)
                )
                val uploadUrl = init.getString("upload_url")
                val fileUrl = init.getString("file_url")
                val c = URL(uploadUrl).openConnection() as HttpURLConnection
                c.requestMethod = "PUT"
                c.doOutput = true
                c.connectTimeout = 30_000
                c.readTimeout = 120_000
                c.setRequestProperty("Content-Type", "image/jpeg")
                c.setFixedLengthStreamingMode(bytes.size)
                c.outputStream.use { it.write(bytes) }
                val code = c.responseCode
                c.disconnect()
                if (code in 200..299) return fileUrl
            } catch (e: FalException) {
                if (e.message?.contains("API key") == true) throw e
            } catch (_: Exception) {
            }
        }
        return "data:image/jpeg;base64," + Base64.encodeToString(bytes, Base64.NO_WRAP)
    }

    class Job(val statusUrl: String, val responseUrl: String)

    fun submit(quality: AiQuality, imageUrl: String, prompt: String, seconds: Int, lockEnd: Boolean): Job {
        val input = JSONObject()
            .put("start_image_url", imageUrl)
            .put("prompt", prompt)
            .put("duration", seconds)
        quality.resolution?.let { input.put("resolution", it) }
        if (lockEnd && quality.endFrame) input.put("end_image_url", imageUrl)
        val r = postJson("https://queue.fal.run/${quality.model}", input)
        val id = r.optString("request_id")
        val status = r.optString("status_url").ifEmpty { "https://queue.fal.run/${quality.model}/requests/$id/status" }
        val response = r.optString("response_url").ifEmpty { "https://queue.fal.run/${quality.model}/requests/$id" }
        return Job(status, response)
    }

    /** Returns "IN_QUEUE", "IN_PROGRESS" or "COMPLETED". */
    fun status(job: Job): String = getJson(job.statusUrl).second?.optString("status") ?: "IN_QUEUE"

    fun resultVideoUrl(job: Job): String {
        val j = getJson(job.responseUrl).second ?: throw FalException("fal.ai returned no result")
        val video = j.optJSONObject("video") ?: throw FalException("fal.ai returned no video: ${j.toString().take(200)}")
        return video.getString("url")
    }

    fun download(url: String, to: File) {
        val tmp = File(to.path + ".part")
        val c = URL(url).openConnection() as HttpURLConnection
        c.connectTimeout = 30_000
        c.readTimeout = 120_000
        if (c.responseCode !in 200..299) throw FalException("Couldn't download the clip (${c.responseCode})")
        c.inputStream.use { input -> tmp.outputStream().use { input.copyTo(it) } }
        c.disconnect()
        if (!tmp.renameTo(to)) throw IOException("Couldn't save the clip")
    }

class GlFrameWriter(surface: Surface, private val w: Int, private val h: Int) {
    private val display: EGLDisplay = EGL14.eglGetDisplay(EGL14.EGL_DEFAULT_DISPLAY)
    private val context: EGLContext
    private val eglSurface: EGLSurface
    private val program: Int
    private val aPos: Int
    private val aTex: Int
    private val uTex: Int
    private val texture: Int
    private var uploaded = false

    private val posBuf = floatBuffer(floatArrayOf(-1f, -1f, 1f, -1f, -1f, 1f, 1f, 1f))
    private val texBuf = floatBuffer(floatArrayOf(0f, 1f, 1f, 1f, 0f, 0f, 1f, 0f))

    init {
        val version = IntArray(2)
        check(EGL14.eglInitialize(display, version, 0, version, 1)) { "eglInitialize failed" }
        val attribs = intArrayOf(
            EGL14.EGL_RED_SIZE, 8,
            EGL14.EGL_GREEN_SIZE, 8,
            EGL14.EGL_BLUE_SIZE, 8,
            EGL14.EGL_ALPHA_SIZE, 8,
            EGL14.EGL_RENDERABLE_TYPE, EGL14.EGL_OPENGL_ES2_BIT,
            EGL_RECORDABLE_ANDROID, 1,
            EGL14.EGL_NONE
        )
        val configs = arrayOfNulls<EGLConfig>(1)
        val num = IntArray(1)
        check(EGL14.eglChooseConfig(display, attribs, 0, configs, 0, 1, num, 0) && num[0] > 0) { "No EGL config" }
        val config = configs[0]!!
        context = EGL14.eglCreateContext(
            display, config, EGL14.EGL_NO_CONTEXT,
            intArrayOf(EGL14.EGL_CONTEXT_CLIENT_VERSION, 2, EGL14.EGL_NONE), 0
        )
        check(context != EGL14.EGL_NO_CONTEXT) { "eglCreateContext failed" }
        eglSurface = EGL14.eglCreateWindowSurface(display, config, surface, intArrayOf(EGL14.EGL_NONE), 0)
        check(eglSurface != EGL14.EGL_NO_SURFACE) { "eglCreateWindowSurface failed" }
        check(EGL14.eglMakeCurrent(display, eglSurface, eglSurface, context)) { "eglMakeCurrent failed" }

        program = buildProgram(
            """
            attribute vec4 aPos;
            attribute vec2 aTex;
            varying vec2 vTex;
            void main() { gl_Position = aPos; vTex = aTex; }
            """.trimIndent(),
            """
            precision mediump float;
            varying vec2 vTex;
            uniform sampler2D uTex;
            void main() { gl_FragColor = texture2D(uTex, vTex); }
            """.trimIndent()
        )
        aPos = GLES20.glGetAttribLocation(program, "aPos")
        aTex = GLES20.glGetAttribLocation(program, "aTex")
        uTex = GLES20.glGetUniformLocation(program, "uTex")

        val ids = IntArray(1)
        GLES20.glGenTextures(1, ids, 0)
        texture = ids[0]
        GLES20.glBindTexture(GLES20.GL_TEXTURE_2D, texture)
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_MIN_FILTER, GLES20.GL_LINEAR)
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_MAG_FILTER, GLES20.GL_LINEAR)
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_WRAP_S, GLES20.GL_CLAMP_TO_EDGE)
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_WRAP_T, GLES20.GL_CLAMP_TO_EDGE)
    }

    fun draw(frame: Bitmap, ptsNs: Long) {
        GLES20.glBindTexture(GLES20.GL_TEXTURE_2D, texture)
        if (!uploaded) {
            GLUtils.texImage2D(GLES20.GL_TEXTURE_2D, 0, frame, 0)
            uploaded = true
        } else {
            GLUtils.texSubImage2D(GLES20.GL_TEXTURE_2D, 0, 0, 0, frame)
        }
        GLES20.glViewport(0, 0, w, h)
        GLES20.glClearColor(0f, 0f, 0f, 1f)
        GLES20.glClear(GLES20.GL_COLOR_BUFFER_BIT)
        GLES20.glUseProgram(program)
        GLES20.glActiveTexture(GLES20.GL_TEXTURE0)
        GLES20.glUniform1i(uTex, 0)
        GLES20.glEnableVertexAttribArray(aPos)
        GLES20.glVertexAttribPointer(aPos, 2, GLES20.GL_FLOAT, false, 0, posBuf)
        GLES20.glEnableVertexAttribArray(aTex)
        GLES20.glVertexAttribPointer(aTex, 2, GLES20.GL_FLOAT, false, 0, texBuf)
        GLES20.glDrawArrays(GLES20.GL_TRIANGLE_STRIP, 0, 4)
        EGLExt.eglPresentationTimeANDROID(display, eglSurface, ptsNs)
        EGL14.eglSwapBuffers(display, eglSurface)
    }

    fun release() {
        runCatching {
            GLES20.glDeleteTextures(1, intArrayOf(texture), 0)
            GLES20.glDeleteProgram(program)
            EGL14.eglMakeCurrent(display, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_CONTEXT)
            EGL14.eglDestroySurface(display, eglSurface)
            EGL14.eglDestroyContext(display, context)
            EGL14.eglReleaseThread()
            EGL14.eglTerminate(display)
        }
    }

    private fun buildProgram(vs: String, fs: String): Int {
        fun compile(type: Int, src: String): Int {
            val s = GLES20.glCreateShader(type)
            GLES20.glShaderSource(s, src)
            GLES20.glCompileShader(s)
            val ok = IntArray(1)
            GLES20.glGetShaderiv(s, GLES20.GL_COMPILE_STATUS, ok, 0)
            check(ok[0] != 0) { "Shader error: " + GLES20.glGetShaderInfoLog(s) }
            return s
        }
        val p = GLES20.glCreateProgram()
        GLES20.glAttachShader(p, compile(GLES20.GL_VERTEX_SHADER, vs))
        GLES20.glAttachShader(p, compile(GLES20.GL_FRAGMENT_SHADER, fs))
        GLES20.glLinkProgram(p)
        val ok = IntArray(1)
        GLES20.glGetProgramiv(p, GLES20.GL_LINK_STATUS, ok, 0)
        check(ok[0] != 0) { "Program error: " + GLES20.glGetProgramInfoLog(p) }
        return p
    }

    companion object {
        private const val EGL_RECORDABLE_ANDROID = 0x3142
        private fun floatBuffer(a: FloatArray): FloatBuffer =
            ByteBuffer.allocateDirect(a.size * 4).order(ByteOrder.nativeOrder()).asFloatBuffer().apply {
                put(a); position(0)
            }
    }
}

    private fun prepareUpload(item: PhotoItem): ByteArray {
        val ratio = if (aspectTag() == "4x5") 4f / 5f else 9f / 16f
        val maxW = 1080
        val src = ImageDecoder.decodeBitmap(ImageDecoder.createSource(contentResolver, item.uri)) { dec, info, _ ->
            dec.allocator = ImageDecoder.ALLOCATOR_SOFTWARE
            var smp = 1
            while (minOf(info.size.width, info.size.height) / (smp * 2) >= maxW * 1.4f) smp *= 2
            if (smp > 1) dec.setTargetSampleSize(smp)
        }
        var cw = src.width.toFloat()
        var ch = cw / ratio
        if (ch > src.height) { ch = src.height.toFloat(); cw = ch * ratio }
        val x = ((src.width - cw) / 2).toInt()
        val y = ((src.height - ch) / 2).toInt()
        val outW = minOf(maxW, cw.toInt())
        val outH = (outW / ratio).toInt()
        val out = Bitmap.createBitmap(outW, outH, Bitmap.Config.ARGB_8888)
        val c = android.graphics.Canvas(out)
        c.drawBitmap(src, android.graphics.Rect(x, y, x + cw.toInt(), y + ch.toInt()),
            android.graphics.Rect(0, 0, outW, outH), android.graphics.Paint(android.graphics.Paint.FILTER_BITMAP_FLAG))
        src.recycle()
        val bos = ByteArrayOutputStream()
        out.compress(Bitmap.CompressFormat.JPEG, 93, bos)
        out.recycle()
        return bos.toByteArray()
    }

enum class ReelFormat(val label: String, val tag: String, val sizes: List<Pair<Int, Int>>) {

fun smooth(x: Float): Float {
    val v = x.coerceIn(0f, 1f); return v * v * (3 - 2 * v)
}

fun easeOut(x: Float): Float {
    val v = x.coerceIn(0f, 1f); val q = 1 - v; return 1 - q * q * q
}

/**
 * Renders the reel on the phone: every frame is drawn with Canvas, handed to the
 * hardware H.264 encoder through OpenGL, and written to an MP4 together with the music.
 */
class ReelRenderer(
    private val ctx: Context,
    private val spec: ReelSpec,
    private val format: ReelFormat,
    private val onProgress: (Float) -> Unit,
) {
    val W: Int
    val H: Int
    private val u: Float
    var warning: String? = null
        private set

    private val fps = 30
    private val xf = 0.5f          // crossfade seconds
    private val endSec = 4.2f      // end card seconds

    init {
        val (w, h) = pickSize()
        W = w; H = h
        u = W / 1080f
    }

    private fun pickSize(): Pair<Int, Int> {
        val encoders = MediaCodecList(MediaCodecList.REGULAR_CODECS).codecInfos.filter { info ->
            info.isEncoder && info.supportedTypes.any { it.equals(MediaFormat.MIMETYPE_VIDEO_AVC, true) }
        }
        for ((w, h) in format.sizes) {
            val ok = encoders.any { info ->
                runCatching {
                    info.getCapabilitiesForType(MediaFormat.MIMETYPE_VIDEO_AVC).videoCapabilities.isSizeSupported(w, h)
                }.getOrDefault(false)
            }
            if (ok) return w to h
        }
        return format.sizes.last()
    }

    // ---------------- timeline ----------------

    private val n get() = spec.slides.size
    // AI clips play for their own length (minus a little at the end); photos use the chosen seconds.
    private val durations: List<Float> = spec.slides.map { s ->
        s.clip?.let { (ClipReader.durationOf(it) - 0.15f).coerceAtLeast(1f) } ?: spec.secondsPerPhoto
    }
    private fun sceneDur(i: Int) = if (i < n) durations[i] else endSec
    private val totalSec get() = durations.sum() + endSec

    private val readers = LinkedHashMap<Int, ClipReader>()

    private fun reader(i: Int): ClipReader {
        readers[i]?.let { return it }
        val r = ClipReader(spec.slides[i].clip!!)
        readers[i] = r
        while (readers.size > 2) {
            val oldest = readers.keys.first()
            readers.remove(oldest)?.release()
        }
        return r
    }

    private fun drawClip(c: Canvas, i: Int, t: Float, dur: Float) {
        val f = reader(i).frameAt(t)
        val zoom = 1f + 0.03f * (t / dur).coerceIn(0f, 1f)
        val s = max(W.toFloat() / f.width, H.toFloat() / f.height) * zoom
        matrix.setScale(s, s)
        matrix.postTranslate((W - f.width * s) / 2f, (H - f.height * s) / 2f)
        c.drawBitmap(f, matrix, bmpPaint)
    }

    // ---------------- photos ----------------

    private val cache = LinkedHashMap<Int, Bitmap>()

    private fun photo(i: Int): Bitmap {
        cache[i]?.let { return it }
        val b = loadPhoto(spec.slides[i].uri)
        cache[i] = b
        while (cache.size > 2) {
            val oldest = cache.keys.first()
            cache.remove(oldest)?.recycle()
        }
        return b
    }

    private fun loadPhoto(uri: Uri): Bitmap {
        val aspect = W.toFloat() / H
        val src = ImageDecoder.createSource(ctx.contentResolver, uri)
        var bmp = ImageDecoder.decodeBitmap(src) { dec, info, _ ->
            dec.allocator = ImageDecoder.ALLOCATOR_SOFTWARE
            val iw = info.size.width.toFloat()
            val ih = info.size.height.toFloat()
            val maxH = max(min(ih, iw / aspect), min(iw, ih / aspect))
            var sample = 1
            while (maxH / (sample * 2) >= 1.25f * H) sample *= 2
            if (sample > 1) dec.setTargetSampleSize(sample)
        }
        // Keep about 1.25x the frame height of detail: sharp when zoomed in, no shimmer when zoomed out.
        val maxH = min(bmp.height.toFloat(), bmp.width / aspect)
        val want = 1.25f * H
        if (maxH > want * 1.3f) {
            val s = want / maxH
            val scaled = Bitmap.createScaledBitmap(bmp, max(1, (bmp.width * s).roundToInt()), max(1, (bmp.height * s).roundToInt()), true)
            if (scaled != bmp) bmp.recycle()
            bmp = scaled
        }
        return bmp
    }

    // zoom start, zoom end, centre start x/y, centre end x/y
    private val motions = arrayOf(
        floatArrayOf(0.95f, 0.72f, 0.50f, 0.50f, 0.42f, 0.44f),
        floatArrayOf(0.76f, 0.95f, 0.40f, 0.42f, 0.50f, 0.50f),
        floatArrayOf(0.92f, 0.68f, 0.50f, 0.50f, 0.57f, 0.46f),
        floatArrayOf(0.86f, 0.86f, 0.36f, 0.42f, 0.64f, 0.58f),
        floatArrayOf(0.72f, 0.95f, 0.57f, 0.46f, 0.50f, 0.50f),
    )

    private val bmpPaint = Paint(Paint.ANTI_ALIAS_FLAG or Paint.FILTER_BITMAP_FLAG)
    private val matrix = Matrix()

    private fun drawPhoto(c: Canvas, bmp: Bitmap, t: Float, dur: Float, m: FloatArray) {
        val lin = (t / dur).coerceIn(0f, 1f)
        val p = smooth(lin) * 0.8f + lin * 0.2f
        val z = m[0] + (m[1] - m[0]) * p
        val cx = m[2] + (m[4] - m[2]) * p
        val cy = m[3] + (m[5] - m[3]) * p
        val iw = bmp.width.toFloat()
        val ih = bmp.height.toFloat()
        val aspect = W.toFloat() / H
        val maxH = min(ih, iw / aspect)
        val ch = maxH * z
        val cw = ch * aspect
        val x0 = (cx * iw - cw / 2).coerceIn(0f, max(0f, iw - cw))
        val y0 = (cy * ih - ch / 2).coerceIn(0f, max(0f, ih - ch))
        val s = W / cw
        matrix.setTranslate(-x0, -y0)
        matrix.postScale(s, s)
        c.drawBitmap(bmp, matrix, bmpPaint)
    }

    // ---------------- captions ----------------

    private val textTop get() = if (format == ReelFormat.REELS) 0.555f * H else 0.76f * H
    private val bandPaint = Paint()
    private val bigPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        typeface = Typeface.create("sans-serif-medium", Typeface.NORMAL)
        textAlign = Paint.Align.CENTER
        letterSpacing = 0.18f
    }
    private val smallPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        typeface = Typeface.create("sans-serif", Typeface.NORMAL)
        textAlign = Paint.Align.CENTER
        letterSpacing = 0.22f
    }

    private fun fitLines(text: String, paint: Paint, baseSize: Float): Pair<List<String>, Float> {
        val maxW = W * 0.86f
        paint.textSize = baseSize
        if (paint.measureText(text) <= maxW) return listOf(text) to baseSize
        val shrunk = baseSize * maxW / paint.measureText(text)
        if (shrunk >= baseSize * 0.78f) return listOf(text) to shrunk
        // two lines, split at the space nearest the middle
        val mid = text.length / 2
        var best = -1
        for (i in text.indices) if (text[i] == ' ' && (best < 0 || kotlin.math.abs(i - mid) < kotlin.math.abs(best - mid))) best = i
        if (best < 0) return listOf(text) to max(shrunk, baseSize * 0.5f)
        val lines = listOf(text.substring(0, best).trim(), text.substring(best + 1).trim())
        paint.textSize = baseSize
        val widest = lines.maxOf { paint.measureText(it) }
        val size = if (widest > maxW) baseSize * maxW / widest else baseSize
        return lines to size
    }

    private fun drawCaption(c: Canvas, slide: Slide, t: Float, tIn: Float, tOut: Float) {
        if (slide.caption.isBlank() && slide.small.isBlank()) return
        val out = 1 - smooth((t - tOut) / 0.35f)
        val kAll = smooth((t - tIn + 0.2f) / 0.5f) * out
        if (kAll <= 0f) return
        val big = if (slide.caption.isNotBlank()) fitLines(slide.caption, bigPaint, 50f * u) else (emptyList<String>() to 0f)
        val blockH = (if (slide.small.isNotBlank()) 62f * u else 0f) + big.first.size * 70f * u
        val top = textTop
        val mid = top + blockH / 2
        val half = (if (format == ReelFormat.REELS) 300f else 250f) * u + blockH / 2
        val a = (0.42f * kAll * 255).roundToInt()
        bandPaint.shader = LinearGradient(
            0f, mid - half, 0f, mid + half,
            intArrayOf(
                Color.argb(0, 34, 20, 28), Color.argb((a * 0.55f).roundToInt(), 34, 20, 28),
                Color.argb(a, 34, 20, 28), Color.argb((a * 0.55f).roundToInt(), 34, 20, 28), Color.argb(0, 34, 20, 28)
            ),
            floatArrayOf(0f, 0.28f, 0.5f, 0.72f, 1f), Shader.TileMode.CLAMP
        )
        c.drawRect(0f, mid - half, W.toFloat(), mid + half, bandPaint)

        var y = top
        var delay = 0f
        val shadow = Color.argb(110, 0, 0, 0)
        if (slide.small.isNotBlank()) {
            val k = easeOut((t - tIn) / 0.6f) * out
            if (k > 0f) {
                smallPaint.textSize = 32f * u
                if (smallPaint.measureText(slide.small) > W * 0.86f) smallPaint.textSize *= W * 0.86f / smallPaint.measureText(slide.small)
                smallPaint.color = Color.argb((255 * k).roundToInt(), 255, 255, 255)
                smallPaint.setShadowLayer(8f * u, 0f, 2f * u, shadow)
                c.drawText(slide.small, W / 2f, y + 32f * u + (1 - k) * 18f * u, smallPaint)
            }
            y += 62f * u
            delay = 0.15f
        }
        if (big.first.isNotEmpty()) {
            val k = easeOut((t - tIn - delay) / 0.6f) * out
            if (k > 0f) {
                bigPaint.textSize = big.second
                bigPaint.color = Color.argb((255 * k).roundToInt(), 255, 255, 255)
                bigPaint.setShadowLayer(10f * u, 0f, 2f * u, shadow)
                for (line in big.first) {
                    c.drawText(line, W / 2f, y + 52f * u + (1 - k) * 18f * u, bigPaint)
                    y += 70f * u
                }
            }
        }
    }

    // ---------------- frames ----------------

    private lateinit var endCard: EndCard

    private fun drawScene(c: Canvas, i: Int, t: Float) {
        if (i >= n) {
            endCard.draw(c, t)
            return
        }
        val dur = sceneDur(i)
        if (spec.slides[i].clip != null) drawClip(c, i, t, dur + xf)
        else drawPhoto(c, photo(i), t, dur + xf, motions[i % motions.size])
        drawCaption(c, spec.slides[i], t, if (i == 0) 0.1f else 0.25f, dur - 0.2f)
    }

    private fun drawFrame(c: Canvas, i: Int, t: Float) {
        if (i > 0 && t < xf) {
            if (i - 1 < n) {
                if (spec.slides[i - 1].clip != null) reader(i - 1) else photo(i - 1)
            }
            drawScene(c, i - 1, sceneDur(i - 1) + t)
            val k = smooth(t / xf)
            c.saveLayerAlpha(0f, 0f, W.toFloat(), H.toFloat(), (k * 255).roundToInt())
            drawScene(c, i, t)
            c.restore()
        } else {
            drawScene(c, i, t)
        }
    }

    // ---------------- export ----------------

    fun render(out: File) {
        endCard = EndCard.create(ctx, spec.endCard, W, H)

        val music: EncodedAudio? = spec.music?.let { uri ->
            try {
                AudioMaker.encodeMusic(ctx, uri, (totalSec * 1_000_000).toLong())
            } catch (e: Exception) {
                warning = "The music file couldn't be used, so this video has no sound."
                null
            }
        }

        val frame = Bitmap.createBitmap(W, H, Bitmap.Config.ARGB_8888)
        val canvas = Canvas(frame)
        val totalFrames = (totalSec * fps).roundToInt()
        val bitrate = (10_000_000L * W * H / (1080L * 1920)).toInt().coerceAtLeast(4_000_000)

        val vf = MediaFormat.createVideoFormat(MediaFormat.MIMETYPE_VIDEO_AVC, W, H).apply {
            setInteger(MediaFormat.KEY_COLOR_FORMAT, MediaCodecInfo.CodecCapabilities.COLOR_FormatSurface)
            setInteger(MediaFormat.KEY_BIT_RATE, bitrate)
            setInteger(MediaFormat.KEY_FRAME_RATE, fps)
            setInteger(MediaFormat.KEY_I_FRAME_INTERVAL, 1)
        }
        val enc = MediaCodec.createEncoderByType(MediaFormat.MIMETYPE_VIDEO_AVC)
        enc.configure(vf, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE)
        val inputSurface = enc.createInputSurface()
        enc.start()
        val gl = GlFrameWriter(inputSurface, W, H)
        val muxer = MediaMuxer(out.absolutePath, MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4)
        val info = MediaCodec.BufferInfo()
        var videoTrack = -1
        var started = false

        fun drain(endOfStream: Boolean) {
            var idle = 0
            while (true) {
                val idx = enc.dequeueOutputBuffer(info, if (endOfStream) 10_000 else 0)
                when {
                    idx == MediaCodec.INFO_TRY_AGAIN_LATER -> {
                        if (!endOfStream) return
                        if (++idle > 500) return
                    }
                    idx == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED -> {
                        videoTrack = muxer.addTrack(enc.outputFormat)
                        val audioTrack = music?.let { muxer.addTrack(it.format) } ?: -1
                        muxer.start()
                        started = true
                        if (music != null && audioTrack >= 0) {
                            val ai = MediaCodec.BufferInfo()
                            for (s in music.samples) {
                                ai.set(0, s.data.size, s.ptsUs, s.flags)
                                muxer.writeSampleData(audioTrack, ByteBuffer.wrap(s.data), ai)
                            }
                        }
                    }
                    idx >= 0 -> {
                        val buf = enc.getOutputBuffer(idx)
                        if ((info.flags and MediaCodec.BUFFER_FLAG_CODEC_CONFIG) != 0) info.size = 0
                        if (info.size > 0 && started && buf != null) {
                            buf.position(info.offset)
                            buf.limit(info.offset + info.size)
                            muxer.writeSampleData(videoTrack, buf, info)
                        }
                        enc.releaseOutputBuffer(idx, false)
                        if ((info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) return
                    }
                }
            }
        }

        try {
            var scene = 0
            var sceneStart = 0f
            for (f in 0 until totalFrames) {
                val t = f.toFloat() / fps
                while (scene < n && t >= sceneStart + sceneDur(scene)) {
                    sceneStart += sceneDur(scene); scene++
                }
                canvas.drawColor(Color.BLACK)
                drawFrame(canvas, scene, t - sceneStart)
                gl.draw(frame, f * 1_000_000_000L / fps)
                drain(false)
                if (f % 5 == 0) onProgress(f.toFloat() / totalFrames)
            }
            enc.signalEndOfInputStream()
            drain(true)
        } finally {
            runCatching { enc.stop() }
            runCatching { enc.release() }
            gl.release()
            inputSurface.release()
            if (started) runCatching { muxer.stop() }
            runCatching { muxer.release() }
            frame.recycle()
            cache.values.forEach { it.recycle() }
            cache.clear()
            readers.values.forEach { it.release() }
            readers.clear()
            endCard.recycle()
        }
        if (!started) throw IllegalStateException("The video encoder didn't produce any output.")
        onProgress(1f)
    }
}
